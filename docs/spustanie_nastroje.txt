ros2 launch waverover runtime_stack.launch.py
correction_mode:=imu/optical_flow
ros2 param set /optical_flow_node debug_publish_image true
ros2 launch waverover runtime_stack.launch.py stack_mode:=wander
ros2 launch waverover runtime_stack.launch.py use_slam:=false

ros2 service call /waverover/switch_to_manual std_srvs/srv/Trigger
ros2 service call /waverover/switch_to_wander std_srvs/srv/Trigger

ros2 param set /motor_hat_node imu_kp 0.30
ros2 param set /motor_hat_node imu_ki 0.05
ros2 param set /motor_hat_node imu_kd 0.01
ros2 param set /motor_hat_node imu_deadband 0.02
ros2 param set /motor_hat_node imu_windup 0.30


ros2 param set /lidar_wander_node threshold_m 0.35
ros2 param set /lidar_wander_node forward_speed 0.20
ros2 param set /lidar_wander_node turn_speed 2.00

ros2 launch waverover ball_follow_standalone.launch.py

ros2 param get /ball_follower debug_image

ros2 launch waver_sim launch_sim.launch.py
